class MainScreen extends StatefulWidget {

class _MainScreenState extends State<MainScreen> {
  int _currentIndex = 0;

  void changeScreen(int index) {
    setState(() {
      _currentIndex = index;
    });
  }

  // Screens list
  final List<Widget> _screens = [
    HomeScreen(onBookNow: () {}),
    const BookingScreen(),
    const GalleryScreen(),
    const ContactScreen(),
  ];

  void initState() {
    super.initState();
    // Initialize HomeScreen with callback
    _screens[0] = HomeScreen(onBookNow: () => changeScreen(1));

  final List<BottomNavigationBarItem> _bottomNavItems = [

      icon: Icon(Icons.calendar_today),

      label: 'GALLERY',
    ),
    const BottomNavigationBarItem(
      icon: Icon(Icons.contact_phone),
      label: 'CONTACT',
    ),
  ];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: _screens[_currentIndex],
      bottomNavigationBar: Container(
        decoration: BoxDecoration(
          gradient: const LinearGradient(
            begin: Alignment.topCenter,

      parent: _welcomeController,

      vsync: this,
      duration: const Duration(milliseconds: 900),
    )..repeat(reverse: true);
    _pulseAnim = Tween<double>(begin: 1.0, end: 1.05).animate(
      CurvedAnimation(parent: _pulseController, curve: Curves.easeInOut),
    );

    // Music note gentle rotation
    _noteController = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 2),
    )..repeat(reverse: true);
    _noteRotate = Tween<double>(begin: -0.08, end: 0.08).animate(
      CurvedAnimation(parent: _noteController, curve: Curves.easeInOut),
    );
    _noteScale = Tween<double>(begin: 0.95, end: 1.1).animate(
      CurvedAnimation(parent: _noteController, curve: Curves.easeInOut),
    );

    // Stagger the section entrances
    _welcomeController.forward();
    _staggerTimer2 = Timer(const Duration(milliseconds: 550), () {
      if (mounted) _aboutController.forward();
    });

    // Auto-scroll media
    _startAutoScroll();
  }

  void _startAutoScroll() {
    _autoScrollTimer = Timer.periodic(const Duration(seconds: 3), (timer) {
      if (!mounted || !_pageController.hasClients) return;

      if (_currentMediaIndex < _mediaItems.length - 1) {
        _pageController.nextPage(
          duration: const Duration(milliseconds: 500),
          curve: Curves.easeInOut,
        );
      } else {
        _pageController.animateToPage(
          0,
          duration: const Duration(milliseconds: 500),
          curve: Curves.easeInOut,
        );
      }
    });
  }

  @override
  void dispose() {
    _autoScrollTimer?.cancel();
    _staggerTimer2?.cancel();
    _welcomeController.dispose();
    _aboutController.dispose();
    _pulseController.dispose();
    _noteController.dispose();
    _pageController.dispose();
    super.dispose();
  }

  Widget _buildMediaCard(Map<String, dynamic> item) {
    final bool isVideo = item.containsKey('video') && item['video'] != null;
    final String mediaPath = isVideo ? item['video'] : item['image'];

    return GestureDetector(
      onTap: () {
        if (isVideo) {
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(content: Text('Video playback feature coming soon!')),
          );
        } else {
          _showImageDialog(mediaPath, item['title']);
        }
      },
      child: Container(
        width: 200, // Fixed width for 3:4 calculation
        margin: const EdgeInsets.symmetric(horizontal: 8, vertical: 8),
        decoration: BoxDecoration(
          borderRadius: BorderRadius.circular(20),
          color: Colors.white,

              blurRadius: 10,
              offset: const Offset(0, 5),

        ),
        child: ClipRRect(
          borderRadius: BorderRadius.circular(20),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Expanded(
                child: AspectRatio(
                  aspectRatio: 3 / 4,

  Future<void> _launchURL(String url) async {
    try {
      final Uri uri = Uri.parse(url);
      if (await canLaunchUrl(uri)) {
        await launchUrl(uri);
      } else {
        throw 'Could not launch $url';
      }
    } catch (e) {
      // Handle error silently
    }
  }
